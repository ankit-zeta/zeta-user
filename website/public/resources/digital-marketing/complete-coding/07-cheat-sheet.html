Quick commands and syntax for daily development tasks

Git Commands

Setup

git init
git remote add origin URL
git branch -M main
git push -u origin main

Branching

git checkout -b feature/name
git switch main
git merge feature/name
git branch -d feature/name

Staging

git add . (all changes)
git add -p (interactive)
git reset HEAD file
git stash / git stash pop

Undoing

git reset --soft HEAD~1
git reset --hard HEAD~1
git revert COMMIT
git checkout -- file

Next.js Commands

Create Project

npx create-next-app@latest
--ts --tailwind --app --src-dir
--import-alias "@/*"

Development

npm run dev (localhost:3000)
npm run build (production)
npm run start (serve build)
npm run lint (check errors)

Routing

app/page.tsx = /
app/about/page.tsx = /about
app/blog/[slug]/page.tsx = /blog/:slug
app/(auth)/login/page.tsx = group

API Routes

app/api/users/route.ts
GET, POST, PUT, PATCH, DELETE
export async function GET(req) {}
NextResponse.json(data)

Prisma Commands

Setup

npx prisma init
npx prisma generate
npx prisma db push (dev)
npx prisma migrate dev

Queries

prisma.user.findMany({})
prisma.user.findUnique({where: {id}})
prisma.user.create({data: {...}})
prisma.user.update({where, data})

Relations

include: { author: true }
include: { posts: true }
include: { _count: { select: { posts: true } } }
select: { name: true, email: true }

Advanced

where: { title: { contains: "q", mode: "insensitive" } }
orderBy: { createdAt: "desc" }
take: 10, skip: 0
prisma.$transaction([...])

TypeScript Patterns

Props Type

type Props = {
  title: string;
  count?: number;
  onClick: () => void;
};

API Response

type ApiResponse = {
  success: boolean;
  data?: T;
  error?: string;
};

Strict Nulls

user?.name ?? "Anonymous"
const arr: string[] = []
arr[0]?.length // might be undefined
use ?? not || for 0 and ""

Utility Types

Partial<User> (all optional)
Required<User> (all required)
Pick<User, "id"|"name">
Omit<User, "password">
