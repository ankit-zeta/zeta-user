Control Flow — Conditionals

JavaScript
if (age >= 18) {
  console.log("Adult");
} else if (age >= 13) {
  console.log("Teen");
} else {
  console.log("Child");
}
Python
if age >= 18:
  print("Adult")
elif age >= 13:
  print("Teen")
else:
  print("Child")

Loops

JavaScript
// For loop
for (let i = 0; i < 5; i++) {
  console.log(i);
}

// For...of
for (const item of items) {
  console.log(item);
}
Python
for i in range(5):
  print(i)

for item in items:
  print(item)

Functions

JavaScript
// Function declaration
function greet(name) {
  return `Hello, ${name}!`;
}

// Arrow function
const greet = (name) => `Hello, ${name}!`;
Python
def greet(name):
  return f"Hello, {name}!"

Arrays & Objects

JavaScript
// Array
const fruits = ["apple", "banana", "cherry"];
fruits.push("date");
fruits.filter(f => f.length > 5);

// Object
const user = { name: "Alice", age: 25 };
user.email = "alice@email.com";
Python
# List
fruits = ["apple", "banana", "cherry"]
fruits.append("date")

# Dictionary
user = { "name": "Alice", "age": 25 }
user["email"] = "alice@email.com"

CSS Selectors

/* Element selector */
p { color: #333; }

/* Class selector */
.highlight { background: yellow; }

/* ID selector */
#header { font-size: 24px; }

/* Descendant */
nav a { text-decoration: none; }

/* Pseudo-class */
a:hover { color: #176B4D; }
